## 二、DNA 真实对接功能设计（P2 优先级）

### 2.1 背景与需求分析

#### 现有痛点
- ❌ 族谱记载的血缘关系存在争议时无法验证
- ❌ 过继/收养等复杂家庭关系难以确认
- ❌ 失联人员回归后缺乏科学认证手段

#### DNA 对接价值
- ✅ Y-DNA 检测父系血脉（验证姓氏传承）
- ✅ mt-DNA 检测母系血脉（追溯外祖父线）
- ✅ autosomal DNA 检测全血缘（确定亲缘等级）
- ✅ Family Tree DNA 数据库匹配同姓族人

## 三、区块链存证功能设计（P3 优先级）

### 3.1 背景与技术选型

#### 为什么需要区块链？

**当前问题**:
- ❌ 电子数据容易被篡改（虽然 audit_logs 可追踪但不够透明）
- ❌ 第三方机构信任缺失（如档案馆损坏丢失）
- ❌ 跨代传承后的数据完整性无法保证

**区块链解决方案**:
- ✅ 时间戳证明：数据在特定时间已存在且未被修改
- ✅ 去中心化存储：分布式节点备份确保持久性
- ✅ 智能合约执行：规则自动化执行（如公示期自动结束）
- ✅ 密码学保证：SHA-256 哈希值防篡改

#### 链上架构选型

| 选项 | 优点 | 缺点 | 适用场景 |
|------|------|------|---------|
| **以太坊主网** | 安全性最高 | Gas 费昂贵 | 高价值数据（祖产契约） |
| **以太坊 Layer2** | 成本低，安全仍强 | TPS 有限 | 常规族谱数据存证 |
| **Polygon PoS** | 超低费用，兼容 EVM | 中心性略高 | 大量小数据存证 |
| **Hyperledger Fabric** | 联盟链模式，可控 | 部署复杂度高 | 多分支协同平台 |
| **国产区块链** | 合规性强 | 生态不完善 | 国内学术机构合作 |

**推荐方案**: Polygon PoS + IPFS 混合存储

#### 智能合约设计（Solidity）

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract HaoGenealogyVerifier {
    
    struct Evidence {
        bytes32 dataHash;      // 数据 SHA-256 哈希
        uint256 timestamp;     // 上链时间戳
        address creator;       // 创建者钱包地址
        string evidenceType;   // "MEMBER" / "RELATION" / "DOCUMENT"
        bool verified;         // 是否已验证
    }
    
    mapping(bytes32 => Evidence) public evidences;
    mapping(address => uint256[]) public userEvidences;
    
    event EvidenceStored(bytes32 indexed dataHash, address creator, uint256 timestamp);
    
    /**
     * 存储成员数据哈希（匿名化版本）
     */
    function storeMemberEvidence(
        bytes32 _dataHash,
        string memory _evidenceType
    ) external returns (bool) {
        require(_dataHash != bytes32(0), "Invalid hash");
        
        Evidence storage evidence = evidences[_dataHash];
        
        // 防止重复存储
        require(!evidence.verified, "Evidence already stored");
        
        evidence.dataHash = _dataHash;
        evidence.timestamp = block.timestamp;
        evidence.creator = msg.sender;
        evidence.evidenceType = _evidenceType;
        evidence.verified = true;
        
        userEvidences[msg.sender].push(evidences.length);
        
        emit EvidenceStored(_dataHash, msg.sender, block.timestamp);
        
        return true;
    }
    
    /**
     * 验证数据是否存在且未被修改
     */
    function verifyEvidence(bytes32 _dataHash) external view returns (bool, uint256) {
        Evidence storage evidence = evidences[_dataHash];
        
        if (!evidence.verified) {
            return (false, 0);
        }
        
        return (true, evidence.timestamp);
    }
    
    /**
     * 批量存储优化（降低 Gas 成本）
     */
    function bulkStoreEvidence(
        bytes32[] memory _dataHashes,
        string[] memory _types
    ) external payable {
        require(_dataHashes.length == _types.length, "Array length mismatch");
        
        for (uint i = 0; i < _dataHashes.length; i++) {
            storeMemberEvidence(_dataHashes[i], _types[i]);
        }
    }
}
```

```javascript
async function verifyMemberEvidence(memberId) {
  // 1. 获取本地数据快照
  const { snapshot, hash } = await generateDataSnapshot(memberId);
```

```javascript
/**
 * 调用 LLM 生成名字建议
 */
export async function generateNameSuggestions(
  birthYear,
  gender,
  generationChar,
  avoidWords = [],
  count = 10
) {
  const prompt = buildNamingPrompt({
    surname: '郝',
    generationChar,
    birthYear,
    gender,
    avoidWords,
    requestCount: count
  });
  
  const response = await openai.createChatCompletion({
    model: 'gpt-4-turbo',
    messages: [{ role: 'user', content: prompt }],
    temperature: 0.7,
    max_tokens: 500
  });
  
  const suggestions = JSON.parse(response.data.choices[0].message.content);
  
  // 二次过滤（重名检测 + 合理性检查）
  const filtered = suggestions.filter(s => 
    !isOverused(s.name) && isValidCharacter(s.name)
  );
  
  return {
    suggestions: filtered.slice(0, count),
    recommended: filtered[0], // AI 推荐的头号人选
    stats: {
      totalGenerated: count,
      filteredOut: count - filtered.length
    }
  };
}

/**
 * 重名检测（实时查询）
 */
async function isOverused(name, branchId) {
  const duplicates = await members.find({
    genealogyName: name,
    branchCode: branchId
  }).limit(3);
  
  return duplicates.length >= 2; // 若已有 2 人以上同名视为重名过多
}
```

```vue
      birthYear: year.value,
      avoidWords: avoidWords.value
    });
    
    suggestions.value = result.suggestions;
    recommended.value = result.recommended;
  } finally {
    loading.value = false;
  }
};

const useAsGenealogyName = (item) => {
  // 将推荐名字填入入谱表单
  entryForm.value.genealogyName = item.name;
  uni.showToast({ title: '已填入谱名字段' });
};
</script>
```
